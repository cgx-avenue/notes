# XGBoost
核心思想是迭代生成树，每次迭代极致梯度提升。

可以进行时序数据的预测，一般而言会有三类特征。

- [日期特征处理](https://www.explinks.com/blog/wx-breaking-through-xgboost-for-time-series-forecasting/#title-6 "日期特征处理")
- [类别特征编码](https://www.explinks.com/blog/wx-breaking-through-xgboost-for-time-series-forecasting/#title-7 "类别特征编码")
- [滞后特征生成](https://www.explinks.com/blog/wx-breaking-through-xgboost-for-time-series-forecasting/#title-8 "滞后特征生成")

别的基本都一致了。

我让deepseek-r1生成的代码，测试可行。
```python
import pandas as pd

import matplotlib.pyplot as plt

from sklearn.preprocessing import LabelEncoder

from xgboost import XGBRegressor

  

# load data and pre prossing

data = pd.read_csv("processed_data/Meas Current LED@DS1 RED.csv")

data['test_start_time_utc'] = pd.to_datetime(data['test_start_time_utc'],format="mixed")

data = data.sort_values('test_start_time_utc').reset_index(drop=True)

data['is_forecast'] = False # add label column and initialized as False

  

# time domain feature

data['year'] = data['test_start_time_utc'].dt.year

data['month'] = data['test_start_time_utc'].dt.month

data['day'] = data['test_start_time_utc'].dt.day

data['hour'] = data['test_start_time_utc'].dt.hour

data['day_of_week'] = data['test_start_time_utc'].dt.dayofweek

  

# label features

label_encoders = {}

for col in ['operator', 'product_part_no']:

le = LabelEncoder()

data[col] = le.fit_transform(data[col])

label_encoders[col] = le

  

# lag features

lags = 3

for i in range(1, lags+1):

data[f'lag_{i}'] = data['attribute_value'].shift(i)

lag_columns = [f'lag_{i}' for i in range(1, lags+1)]

# fill with 0

data[lag_columns] = data[lag_columns].fillna(0)

# fill NA with 0

data=data.fillna(0)

# filter rows where attribute_value=0

data = data[data["attribute_value"] != 0]

  

# Model training

features = ['year', 'month', 'day', 'hour', 'day_of_week',

'operator', 'product_part_no', 'lag_1', 'lag_2', 'lag_3']

X = data[features]

y = data['attribute_value']

split_idx = int(len(X) * 0.8)

X_train, X_test = X[:split_idx], X[split_idx:]

y_train, y_test = y[:split_idx], y[split_idx:]

  

model = XGBRegressor()

model.fit(X_train, y_train)

  

# 预测未来10个点并追加到DataFrame（间隔20秒）

last_row = data.iloc[-1].copy()

future_predictions = []

  

for _ in range(10):

next_time = last_row['test_start_time_utc'] + pd.Timedelta(seconds=20)

next_value = model.predict(last_row[features].values.reshape(1, -1))[0]

future_predictions.append(next_value)

# 创建新行并标记为预测点

new_row = last_row.copy()

new_row['test_start_time_utc'] = next_time

new_row['attribute_value'] = next_value

new_row['is_forecast'] = True # 标记为预测点

# 更新滞后特征和时间特征

new_row['lag_3'] = new_row['lag_2']

new_row['lag_2'] = new_row['lag_1']

new_row['lag_1'] = next_value

new_row['year'] = next_time.year

new_row['month'] = next_time.month

new_row['day'] = next_time.day

new_row['hour'] = next_time.hour

new_row['day_of_week'] = next_time.dayofweek

# 追加到DataFrame

data = pd.concat([data, pd.DataFrame([new_row])], ignore_index=True)

last_row = new_row

  

# 绘制最后100个点的时序散点图

last_100 = data.tail(100)

colors = last_100['is_forecast'].map({True: 'red', False: 'blue'})

  

plt.figure(figsize=(15, 6))

plt.scatter(

last_100['test_start_time_utc'],

last_100['attribute_value'],

c=colors,

s=50, # 点大小

alpha=0.8 # 透明度

)

plt.title("Last 100 Values: Historical (Blue) vs Forecast (Red)")

plt.xlabel("Time (UTC)")

plt.ylabel("Attribute Value (mA)")

plt.grid(True, linestyle='--', alpha=0.5)

plt.xticks(rotation=45)

plt.tight_layout()

plt.show()
```
