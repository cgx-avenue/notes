就用下面的链接和yaml配置，用ContainerStation的应用新建，然后复制进去。

[NAS成股神了？部署 AI 炒股，每日自动量化分析+推送提醒 - 少数派](https://sspai.com/post/106821)
```bash
services:
  analyzer:
    image: rcxi27eq7hzj0pedwl.xuanyuan.run/birdxs/daily_stock_analysis:latest
    container_name: dsa-analyzer
    restart: always
    env_file:
      - /share/Container/daily_stock_analysis/.env
    volumes:
      - /share/Container/daily_stock_analysis/data:/app/data
      - /share/Container/daily_stock_analysis/logs:/app/logs
      - /share/Container/daily_stock_analysis/reports:/app/reports
      - /share/Container/daily_stock_analysis/.env:/app/.env
      - /share/Container/daily_stock_analysis/strategies:/app/strategies
    environment:
      - TZ=Asia/Shanghai

  server:
    image: rcxi27eq7hzj0pedwl.xuanyuan.run/birdxs/daily_stock_analysis:latest
    container_name: dsa-server
    restart: always
    env_file:
      - /share/Container/dsa/.env
    environment:
      - TZ=Asia/Shanghai
      - WEBUI_HOST=0.0.0.0
      - WEBUI_PORT=8068
    command: ["python", "main.py", "--serve-only", "--host", "0.0.0.0", "--port", "8068"]
    ports:
      - "8068:8068"
    volumes:
      - /share/Container/daily_stock_analysis/data:/app/data
      - /share/Container/daily_stock_analysis/logs:/app/logs
      - /share/Container/daily_stock_analysis/reports:/app/reports
      - /share/Container/daily_stock_analysis/.env:/app/.env
      - /share/Container/daily_stock_analysis/strategies:/app/strategies

```
