# docker network test using scapy
## environment setup
```bash
# create new bridge network
sudo docker network create brtest
# start 2 ubuntu containers using bridged network, 1 in priviledge mode
sudo docker run -it --network=brtest --name ubuntu_br --restart=always ubuntu
# open another terminal
sudo docker run -it --network=brtest --name ubuntu_brp --privileged ubuntu
# install dependencies on both containers
sed -i s@/archive.ubuntu.com/@/mirrors.aliyun.com/@g /etc/apt/sources.list 
sed -i s@/security.ubuntu.com/@/mirrors.aliyun.com/@g /etc/apt/sources.list  
apt-get clean  
apt-get update
apt install python3 python3-pip python3-scapy -y

```
